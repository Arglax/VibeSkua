using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using Newtonsoft.Json;
using Newtonsoft.Json.Linq;
using Skua.Core.Interfaces;
using Skua.Core.Models;
using Skua.Core.Models.Items;

// Generated by MoVibeSkua before each run (MoVibeSkua.Android/Bot/AutoEnhance.cs). The two methods
// between the AUQW markers are copied word for word from AUQW's CoreAdvanced.cs (SmartEnhance's
// forge and awe libraries), so the recommended sets follow AUQW's updates.
public class MvsAutoEnhance
{
    private IScriptInterface Bot => IScriptInterface.Instance;
    private CoreBots Core => CoreBots.Instance;
    private static CoreAdvanced Adv => CoreAdvanced.Instance;

    // Read by the copied AUQW library methods.
    private string className = "";
    private EnhancementType? type = null;
    private CapeSpecial cSpecial = CapeSpecial.None;
    private HelmSpecial hSpecial = HelmSpecial.None;
    private WeaponSpecial wSpecial = WeaponSpecial.None;

    private const string ConfigFile = "mvs-autoenhance.json";
    private const string ResultFile = "mvs-autoenhance-result.json";
    private const string UltraFile = "mvs-enhance-ultra.json";

    public void ScriptMain(IScriptInterface bot)
    {
        Result result = new();
        try
        {
            Run(result);
        }
        catch (Exception ex)
        {
            result.Status = "error";
            result.Lines.Add("Auto Enhance hit a problem: " + ex.Message);
        }
        try
        {
            File.WriteAllText(Path.Combine(ClientFileSources.SkuaOptionsDIR, ResultFile), JsonConvert.SerializeObject(result));
        }
        catch { }
        foreach (string line in result.Lines)
            Core.Logger(line, "MVS Auto Enhance");
    }

    private class Result
    {
        public string Status = "ok";        // ok, enhanced, partial, failed, error, skipped
        public string Class = "";
        public string UseCase = "";
        public int Before;                   // % of the recommended set equipped before
        public int After;
        public List<string> Lines = new();
        public List<string> Problems = new();
    }

    private class Want
    {
        public EnhancementType Type;
        public WeaponSpecial Weapon;
        public HelmSpecial Helm;
        public CapeSpecial Cape;
        public string Source = "AUQW";
    }

    private void Run(Result result)
    {
        string useCase = "Bossing";
        int threshold = 60;
        try
        {
            JObject cfg = JObject.Parse(File.ReadAllText(Path.Combine(ClientFileSources.SkuaOptionsDIR, ConfigFile)));
            useCase = (string?)cfg["useCase"] ?? useCase;
            threshold = (int?)cfg["threshold"] ?? threshold;
        }
        catch { }
        result.UseCase = useCase;

        InventoryItem? cls = Bot.Inventory.Items.FirstOrDefault(i => i.Equipped && i.Category == ItemCategory.Class);
        if (cls == null)
        {
            result.Status = "skipped";
            result.Lines.Add("No class equipped, nothing to check.");
            return;
        }
        result.Class = cls.Name;

        Want want = Recommended(cls.Name, useCase);
        int before = Score(want, out List<string> wrong, out bool critical);
        result.Before = before;
        string set = Describe(want);
        if (before >= 100)
        {
            result.Lines.Add($"{cls.Name}: already on the recommended {useCase.ToLower()} set ({set}).");
            return;
        }
        if (!critical && before >= threshold)
        {
            result.Lines.Add($"{cls.Name}: {before}% of the recommended set ({set}), above {threshold}%, so left as is. Different: {string.Join(", ", wrong)}.");
            return;
        }

        result.Lines.Add($"{cls.Name}: {before}% of the recommended {useCase.ToLower()} set ({set})" +
                         (critical ? ", and the helm or cape is off" : "") + ". Enhancing: " + string.Join(", ", wrong) + ".");
        int goldBefore = Bot.Player.Gold;
        if (Bot.Player.InCombat)
            Core.JumpWait();
        Adv.EnhanceEquipped(want.Type, want.Cape, want.Helm, want.Weapon, true);
        Bot.Wait.ForTrue(() => false, 6);

        int after = Score(want, out List<string> still, out bool stillCritical);
        result.After = after;
        if (after >= 100 || (!stillCritical && after >= threshold && still.Count == 0))
        {
            result.Status = "enhanced";
            result.Lines.Add($"Done: now {after}% of the set (spent {Math.Max(0, goldBefore - Bot.Player.Gold):N0} gold).");
            return;
        }
        result.Status = after > before ? "partial" : "failed";
        foreach (string slot in still)
            result.Problems.Add(slot + ": " + WhyNot(slot, want));
        result.Lines.Add($"Couldn't finish: now {after}% of the set. " + string.Join(" ", result.Problems));
    }

    // AUQW's pick for the class, then for Ultra the r/AQW Simplified Bosses Guide's set where it has
    // one (each special only when it's unlocked; otherwise AUQW's pick for that slot stays).
    private Want Recommended(string name, string useCase)
    {
        className = name.ToLower().Trim();
        type = null;
        cSpecial = CapeSpecial.None;
        hSpecial = HelmSpecial.None;
        wSpecial = WeaponSpecial.None;
        if (!ForgeEnhancementLibrary())
            AweEnhancementLibrary();
        Want want = new() { Type = type ?? EnhancementType.Lucky, Weapon = wSpecial, Helm = hSpecial, Cape = cSpecial };

        if (useCase == "Ultra")
        {
            try
            {
                JObject ultra = JObject.Parse(File.ReadAllText(Path.Combine(ClientFileSources.SkuaScriptsDIR, "MVS", UltraFile)));
                if (ultra["classes"]?[className] is JObject u)
                {
                    want.Source = "r/AQW Simplified Bosses Guide";
                    if (Enum.TryParse((string?)u["type"], out EnhancementType t))
                        want.Type = t;
                    if (Enum.TryParse((string?)u["weapon"], out WeaponSpecial w) && w != WeaponSpecial.None && Unlocked(w.ToString()))
                        want.Weapon = w;
                    if (Enum.TryParse((string?)u["helm"], out HelmSpecial h) && h != HelmSpecial.None && Unlocked("Helm" + h))
                        want.Helm = h;
                    if (Enum.TryParse((string?)u["cape"], out CapeSpecial c) && c != CapeSpecial.None && Unlocked("Cape" + c))
                        want.Cape = c;
                }
            }
            catch { }
        }
        return want;
    }

    private static readonly Dictionary<string, string> UnlockOf = new()
    {
        ["Dauntless"] = "uDauntless", ["Ravenous"] = "uRavenous", ["Valiance"] = "uValiance", ["Praxis"] = "uPraxis",
        ["Arcanas_Concerto"] = "uArcanasConcerto", ["Lacerate"] = "uLacerate", ["Smite"] = "uSmite",
        ["Elysium"] = "uElysium", ["Acheron"] = "uAcheron", ["Forge"] = "uForgeWeapon",
        ["Awe_Blast"] = "uAwe", ["Health_Vamp"] = "uAwe", ["Mana_Vamp"] = "uAwe", ["Spiral_Carve"] = "uAwe", ["Powerword_Die"] = "uAwe",
        ["HelmAnima"] = "uAnima", ["HelmExamen"] = "uExamen", ["HelmPneuma"] = "uPneuma", ["HelmVim"] = "uVim",
        ["HelmHearty"] = "uHearty", ["HelmForge"] = "uForgeHelm",
        ["CapeVainglory"] = "uVainglory", ["CapeLament"] = "uLament", ["CapeAbsolution"] = "uAbsolution",
        ["CapePenitence"] = "uPenitence", ["CapeAvarice"] = "uAvarice", ["CapeForge"] = "uForgeCape",
    };

    private static bool Unlocked(string key)
    {
        if (!UnlockOf.TryGetValue(key, out string? method))
            return true;
        try
        {
            return (bool)(typeof(CoreAdvanced).GetMethod(method)?.Invoke(Adv, null) ?? true);
        }
        catch
        {
            return true;
        }
    }

    // Weighted match: class enhancement 1, weapon 1, helm 2, cape 2 (helm and cape matter most:
    // a helm can add or cost a lot of HP, a cape gives big utility). A wrong helm or cape is critical.
    private int Score(Want want, out List<string> wrong, out bool critical)
    {
        wrong = new();
        critical = false;
        int total = 0, got = 0;
        InventoryItem? cls = Bot.Inventory.Items.FirstOrDefault(i => i.Equipped && i.Category == ItemCategory.Class);
        bool hasWeapon = Bot.Inventory.Items.Any(i => i.Equipped && i.ItemGroup == "Weapon");
        bool hasHelm = Bot.Inventory.Items.Any(i => i.Equipped && i.Category == ItemCategory.Helm);
        bool hasCape = Bot.Inventory.Items.Any(i => i.Equipped && i.Category == ItemCategory.Cape);

        total += 1;
        if (cls != null && Adv.CurrentClassEnh() == want.Type) got += 1; else wrong.Add($"class ({want.Type})");

        if (hasWeapon)
        {
            total += 1;
            if (want.Weapon == WeaponSpecial.None || Adv.CurrentWeaponSpecial() == want.Weapon) got += 1;
            else wrong.Add($"weapon ({Pretty(want.Weapon.ToString())})");
        }
        if (hasHelm)
        {
            total += 2;
            if (want.Helm == HelmSpecial.None || Adv.CurrentHelmSpecial() == want.Helm) got += 2;
            else { wrong.Add($"helm ({want.Helm})"); critical = true; }
        }
        else if (want.Helm != HelmSpecial.None)
            wrong.Add("helm (none equipped)");
        if (hasCape)
        {
            total += 2;
            if (want.Cape == CapeSpecial.None || Adv.CurrentCapeSpecial() == want.Cape) got += 2;
            else { wrong.Add($"cape ({want.Cape})"); critical = true; }
        }
        else if (want.Cape != CapeSpecial.None)
            wrong.Add("cape (none equipped)");
        return total == 0 ? 100 : (int)Math.Round(100.0 * got / total);
    }

    private string WhyNot(string slot, Want want)
    {
        if (slot.Contains("none equipped"))
            return "equip one to enhance it.";
        string key = slot.StartsWith("helm") ? "Helm" + want.Helm : slot.StartsWith("cape") ? "Cape" + want.Cape
            : slot.StartsWith("weapon") ? want.Weapon.ToString() : "";
        if (key != "" && !Unlocked(key))
            return "that enhancement is still locked (finish its Forge quest).";
        if (Bot.Player.Gold < 1_000_000)
            return $"probably not enough gold ({Bot.Player.Gold:N0} left).";
        return "the shop didn't sell it at your level, or the item can't take it.";
    }

    private static string Describe(Want w) =>
        $"{Pretty(w.Weapon.ToString())} | {w.Type} / {w.Helm} | {w.Cape}".Replace("None", "-");

    private static string Pretty(string s) => s.Replace('_', ' ');

    // ===== AUQW CoreAdvanced.cs SmartEnhance libraries (copied by MVS) =====
    //@@AUQW_LIBRARIES@@
    // ===== end of AUQW copy =====
}
