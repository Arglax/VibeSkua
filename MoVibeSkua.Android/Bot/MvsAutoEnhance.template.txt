using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using Newtonsoft.Json;
using Newtonsoft.Json.Linq;
using Skua.Core.Interfaces;
using Skua.Core.Models;
using Skua.Core.Models.Items;

// Generated by MoVibeSkua before each run (MoVibeSkua.Android/Bot/AutoEnhance.cs). The two methods
// between the AUQW markers are copied word for word from AUQW's CoreAdvanced.cs (SmartEnhance's
// forge and awe libraries), so "best" follows AUQW's updates.
public class MvsAutoEnhance
{
    private IScriptInterface Bot => IScriptInterface.Instance;
    private CoreBots Core => CoreBots.Instance;
    private static CoreAdvanced Adv => CoreAdvanced.Instance;

    // Read by the copied AUQW library methods.
    private string className = "";
    private EnhancementType? type = null;
    private CapeSpecial cSpecial = CapeSpecial.None;
    private HelmSpecial hSpecial = HelmSpecial.None;
    private WeaponSpecial wSpecial = WeaponSpecial.None;

    // The pieces the player ticked; the others are never touched.
    private HashSet<string> slots = new() { "class", "weapon", "helm", "cape" };

    private const string ConfigFile = "mvs-autoenhance.json";
    private const string ResultFile = "mvs-autoenhance-result.json";
    private const string UltraFile = "mvs-enhance-ultra.json";

    public void ScriptMain(IScriptInterface bot)
    {
        Result result = new();
        try
        {
            Run(result);
        }
        catch (Exception ex)
        {
            result.Status = "error";
            result.Lines.Add("Use Best Enhancement hit a problem: " + ex.Message);
        }
        try
        {
            File.WriteAllText(Path.Combine(ClientFileSources.SkuaOptionsDIR, ResultFile), JsonConvert.SerializeObject(result));
        }
        catch { }
        foreach (string line in result.Lines)
            Core.Logger(line, "MVS Use Best Enhancement");
    }

    private class Result
    {
        public string Status = "ok";        // preview, ok (nothing to do), enhanced, partial, failed, error, skipped
        public string Class = "";
        public string UseCase = "";
        public string Source = "";
        public List<Row> Rows = new();
        public List<string> Lines = new();
        public List<string> Problems = new();
    }

    private class Row
    {
        public string Slot = "";          // class, weapon, helm, cape
        public string Item = "";          // equipped item, "" if none
        public string Now = "";
        public string Best = "";
        public bool Ok;                   // already on the best enhancement
        public bool Locked;               // the best enhancement isn't unlocked yet
    }

    private class Want
    {
        public EnhancementType Type;
        public WeaponSpecial Weapon;
        public HelmSpecial Helm;
        public CapeSpecial Cape;
        public string Source = "Skua script team (AUQW SmartEnhance)";
    }

    private void Run(Result result)
    {
        string useCase = "Bossing";
        bool preview = false;
        try
        {
            JObject cfg = JObject.Parse(File.ReadAllText(Path.Combine(ClientFileSources.SkuaOptionsDIR, ConfigFile)));
            useCase = (string?)cfg["useCase"] ?? useCase;
            preview = (bool?)cfg["preview"] ?? false;
            if (cfg["slots"] is JArray arr)
                slots = new HashSet<string>(arr.Select(t => ((string?)t ?? "").ToLower()));
        }
        catch { }
        result.UseCase = useCase;

        InventoryItem? cls = Equipped("class");
        if (cls == null)
        {
            result.Status = "skipped";
            result.Lines.Add("No class equipped, nothing to check.");
            return;
        }
        result.Class = cls.Name;
        Want want = Recommended(cls.Name, useCase);
        result.Source = want.Source;
        result.Rows = Rows(want);

        string summary = string.Join(", ", result.Rows.Where(r => r.Item != "").Select(r => $"{r.Slot} {(r.Ok ? "ok" : r.Now + " -> " + r.Best)}"));
        if (preview)
        {
            result.Status = "preview";
            result.Lines.Add($"{cls.Name} ({useCase.ToLower()}): {summary}.");
            return;
        }

        List<Row> todo = result.Rows.Where(r => slots.Contains(r.Slot) && r.Item != "" && !r.Ok).ToList();
        if (todo.Count == 0)
        {
            result.Status = "ok";
            result.Lines.Add($"{cls.Name}: every ticked piece is already on its best {useCase.ToLower()} enhancement.");
            return;
        }

        result.Lines.Add($"{cls.Name} ({useCase.ToLower()}): enhancing " + string.Join(", ", todo.Select(r => $"{r.Slot} {r.Item} ({r.Now} -> {r.Best})")) + ".");
        int goldBefore = Bot.Player.Gold;
        if (Bot.Player.InCombat)
            Core.JumpWait();
        foreach (Row row in todo)
        {
            if (row.Locked)
                continue;
            try
            {
                switch (row.Slot)
                {
                    case "class": Adv.EnhanceItem(row.Item, want.Type); break;
                    case "weapon": Adv.EnhanceItem(row.Item, want.Type, wSpecial: want.Weapon); break;
                    case "helm": Adv.EnhanceItem(row.Item, want.Type, hSpecial: want.Helm); break;
                    case "cape": Adv.EnhanceItem(row.Item, want.Type, cSpecial: want.Cape); break;
                }
            }
            catch (Exception ex)
            {
                result.Problems.Add($"{row.Slot}: {ex.Message}");
            }
        }
        Bot.Wait.ForTrue(() => false, 6);

        result.Rows = Rows(want);
        List<Row> still = result.Rows.Where(r => slots.Contains(r.Slot) && r.Item != "" && !r.Ok).ToList();
        int spent = Math.Max(0, goldBefore - Bot.Player.Gold);
        if (still.Count == 0)
        {
            result.Status = "enhanced";
            result.Lines.Add($"Done: every ticked piece is on its best enhancement (spent {spent:N0} gold).");
            return;
        }
        result.Status = still.Count < todo.Count ? "partial" : "failed";
        foreach (Row row in still)
            result.Problems.Add($"{row.Slot} ({row.Item}): {WhyNot(row)}");
        result.Lines.Add($"Couldn't finish {still.Count} of {todo.Count} (spent {spent:N0} gold). " + string.Join(" ", result.Problems));
    }

    private InventoryItem? Equipped(string slot) => slot switch
    {
        "class" => Bot.Inventory.Items.FirstOrDefault(i => i.Equipped && i.Category == ItemCategory.Class),
        "weapon" => Bot.Inventory.Items.FirstOrDefault(i => i.Equipped && i.ItemGroup == "Weapon"),
        "helm" => Bot.Inventory.Items.FirstOrDefault(i => i.Equipped && i.Category == ItemCategory.Helm),
        "cape" => Bot.Inventory.Items.FirstOrDefault(i => i.Equipped && i.Category == ItemCategory.Cape),
        _ => null
    };

    // One row per piece: what it has now, what's best, whether it already matches.
    private List<Row> Rows(Want want)
    {
        List<Row> rows = new();
        foreach (string slot in new[] { "class", "weapon", "helm", "cape" })
        {
            InventoryItem? item = Equipped(slot);
            Row row = new() { Slot = slot, Item = item?.Name ?? "" };
            string type = want.Type.ToString();
            switch (slot)
            {
                case "class":
                    row.Best = type;
                    row.Ok = item != null && SameType(item, type);
                    break;
                case "weapon":
                    // Forge weapon specials (Lacerate and up: Valiance, Ravenous...) are the whole
                    // enhancement; only the Awe ones (Spiral Carve to Powerword Die) sit on a type.
                    if (want.Weapon == WeaponSpecial.Forge)
                        row.Best = "Forge";
                    else if ((int)want.Weapon > 6)
                        row.Best = Pretty(want.Weapon.ToString());
                    else
                        row.Best = want.Weapon == WeaponSpecial.None ? type : $"{type}, {Pretty(want.Weapon.ToString())}";
                    row.Ok = item != null && (want.Weapon == WeaponSpecial.Forge ? MvsWeaponSpecial() == WeaponSpecial.Forge
                        : (int)want.Weapon > 6 ? MvsWeaponSpecial() == want.Weapon
                        : SameType(item, type) && (want.Weapon == WeaponSpecial.None || MvsWeaponSpecial() == want.Weapon));
                    row.Locked = want.Weapon != WeaponSpecial.None && !Unlocked(want.Weapon.ToString());
                    break;
                case "helm":
                    row.Best = want.Helm == HelmSpecial.None ? type : Pretty(want.Helm.ToString());
                    row.Ok = item != null && (want.Helm == HelmSpecial.None ? SameType(item, type) : Adv.CurrentHelmSpecial() == want.Helm);
                    row.Locked = want.Helm != HelmSpecial.None && !Unlocked("Helm" + want.Helm);
                    break;
                case "cape":
                    row.Best = want.Cape == CapeSpecial.None ? type : Pretty(want.Cape.ToString());
                    row.Ok = item != null && (want.Cape == CapeSpecial.None ? SameType(item, type) : Adv.CurrentCapeSpecial() == want.Cape);
                    row.Locked = want.Cape != CapeSpecial.None && !Unlocked("Cape" + want.Cape);
                    break;
            }
            // AUQW's own test (which also wants the enhancement at your level): "already best" agrees
            // with what AUQW's enhancing would do. The checks above stay as the fallback.
            if (item != null && AuqwHas(item, want, slot) is bool has)
                row.Ok = has;
            row.Now = item == null ? "nothing equipped" : Describe(item)
                + (item.EnhancementLevel > 0 && item.EnhancementLevel < Bot.Player.Level ? $" (level {item.EnhancementLevel})" : "");
            rows.Add(row);
        }
        return rows;
    }

    private static readonly System.Reflection.MethodInfo? AuqwCheck = typeof(CoreAdvanced).GetMethod("AlreadyHasRequestedEnhancement",
        System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Public);

    private bool? AuqwHas(InventoryItem item, Want want, string slot)
    {
        try
        {
            if (AuqwCheck == null || AuqwCheck.GetParameters().Length != 6)
                return null;
            return (bool?)AuqwCheck.Invoke(Adv, new object[] { item, want.Type,
                slot == "cape" ? want.Cape : CapeSpecial.None, slot == "helm" ? want.Helm : HelmSpecial.None,
                slot == "weapon" ? want.Weapon : WeaponSpecial.None, false });
        }
        catch
        {
            return null;
        }
    }

    private static bool SameType(InventoryItem item, string type) =>
        item.EnhancementLevel > 0 && string.Equals(PatternName(item.EnhancementPatternID), type, StringComparison.OrdinalIgnoreCase);

    private static string Describe(InventoryItem item)
    {
        if (item.EnhancementLevel <= 0 && item.EnhancementPatternID <= 0)
            return "not enhanced";
        string pattern = PatternName(item.EnhancementPatternID);
        string? proc = ProcName(item.ProcID);
        // A forge weapon special is the whole enhancement (its pattern reads "Forge").
        return proc == null ? pattern : item.ProcID > 6 ? proc : $"{pattern}, {proc}";
    }

    private static string PatternName(int id) => id switch
    {
        1 => "Adventurer", 2 => "Fighter", 3 => "Thief", 4 => "Armsman", 5 => "Hybrid", 6 => "Wizard",
        7 => "Healer", 8 => "SpellBreaker", 9 => "Lucky", 10 => "Forge", 11 => "Absolution", 12 => "Avarice",
        23 => "Depths", 24 => "Vainglory", 25 => "Vim", 26 => "Examen", 27 => "Pneuma", 28 => "Anima",
        29 => "Penitence", 30 => "Lament", 32 => "Hearty", _ => "Unknown"
    };

    private static string? ProcName(int id) => id switch
    {
        2 => "Spiral Carve", 3 => "Awe Blast", 4 => "Health Vamp", 5 => "Mana Vamp", 6 => "Powerword Die",
        7 => "Lacerate", 8 => "Smite", 9 => "Valiance", 10 => "Arcanas Concerto", 11 => "Acheron",
        12 => "Elysium", 13 => "Praxis", 14 => "Dauntless", 15 => "Ravenous", _ => null
    };

    // AUQW's pick for the class, then for Ultra the r/AQW Simplified Bosses Guide's set where it has
    // one (each special only when it's unlocked; otherwise AUQW's pick for that slot stays).
    private Want Recommended(string name, string useCase)
    {
        className = name.ToLower().Trim();
        type = null;
        cSpecial = CapeSpecial.None;
        hSpecial = HelmSpecial.None;
        wSpecial = WeaponSpecial.None;
        if (!ForgeEnhancementLibrary())
            AweEnhancementLibrary();
        Want want = new() { Type = type ?? EnhancementType.Lucky, Weapon = wSpecial, Helm = hSpecial, Cape = cSpecial };

        if (useCase == "Ultra")
        {
            try
            {
                JObject ultra = JObject.Parse(File.ReadAllText(Path.Combine(ClientFileSources.SkuaScriptsDIR, "MVS", UltraFile)));
                if (ultra["classes"]?[className] is JObject u)
                {
                    want.Source = "r/AQW Simplified Bosses Guide";
                    if (Enum.TryParse((string?)u["type"], out EnhancementType t))
                        want.Type = t;
                    if (Enum.TryParse((string?)u["weapon"], out WeaponSpecial w) && w != WeaponSpecial.None && Unlocked(w.ToString()))
                        want.Weapon = w;
                    if (Enum.TryParse((string?)u["helm"], out HelmSpecial h) && h != HelmSpecial.None && Unlocked("Helm" + h))
                        want.Helm = h;
                    if (Enum.TryParse((string?)u["cape"], out CapeSpecial c) && c != CapeSpecial.None && Unlocked("Cape" + c))
                        want.Cape = c;
                }
            }
            catch { }
        }
        return want;
    }

    private static readonly Dictionary<string, string> UnlockOf = new()
    {
        ["Dauntless"] = "uDauntless", ["Ravenous"] = "uRavenous", ["Valiance"] = "uValiance", ["Praxis"] = "uPraxis",
        ["Arcanas_Concerto"] = "uArcanasConcerto", ["Lacerate"] = "uLacerate", ["Smite"] = "uSmite",
        ["Elysium"] = "uElysium", ["Acheron"] = "uAcheron", ["Forge"] = "uForgeWeapon",
        ["Awe_Blast"] = "uAwe", ["Health_Vamp"] = "uAwe", ["Mana_Vamp"] = "uAwe", ["Spiral_Carve"] = "uAwe", ["Powerword_Die"] = "uAwe",
        ["HelmAnima"] = "uAnima", ["HelmExamen"] = "uExamen", ["HelmPneuma"] = "uPneuma", ["HelmVim"] = "uVim",
        ["HelmHearty"] = "uHearty", ["HelmForge"] = "uForgeHelm",
        ["CapeVainglory"] = "uVainglory", ["CapeLament"] = "uLament", ["CapeAbsolution"] = "uAbsolution",
        ["CapePenitence"] = "uPenitence", ["CapeAvarice"] = "uAvarice", ["CapeForge"] = "uForgeCape",
    };

    private static bool Unlocked(string key)
    {
        if (!UnlockOf.TryGetValue(key, out string? method))
            return true;
        try
        {
            return (bool)(typeof(CoreAdvanced).GetMethod(method)?.Invoke(Adv, null) ?? true);
        }
        catch
        {
            return true;
        }
    }

    private string WhyNot(Row row)
    {
        if (row.Locked)
            return $"{row.Best} is still locked (finish its Forge quest).";
        if (Bot.Player.Gold < 1_000_000)
            return $"probably not enough gold ({Bot.Player.Gold:N0} left).";
        return "the shop didn't sell it at your level, or the item can't take it.";
    }

    // The weapon's special is its proc (ProcID); AUQW's CurrentWeaponSpecial() reads the enhancement
    // pattern instead (10 for any forge weapon), so a Ravenous sword read as Arcana's Concerto. Same
    // rule as AUQW's WeaponAlreadyHasRequestedEnhancement: proc 0 on a forge pattern is Forge.
    private WeaponSpecial MvsWeaponSpecial()
    {
        InventoryItem? w = Bot.Inventory.Items.FirstOrDefault(i => i.Equipped && i.ItemGroup == "Weapon");
        if (w == null || w.EnhancementLevel <= 0)
            return WeaponSpecial.None;
        if (w.ProcID == 0)
            return w.EnhancementPatternID == 10 ? WeaponSpecial.Forge : WeaponSpecial.None;
        return w.ProcID >= 2 && Enum.IsDefined(typeof(WeaponSpecial), w.ProcID) ? (WeaponSpecial)w.ProcID : WeaponSpecial.None;
    }

    private static string Pretty(string s) => s.Replace('_', ' ');

    // ===== AUQW CoreAdvanced.cs SmartEnhance libraries (copied by MVS) =====
    //@@AUQW_LIBRARIES@@
    // ===== end of AUQW copy =====
}
